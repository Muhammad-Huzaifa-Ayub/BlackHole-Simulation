// ── Physical constants (rs = 1 throughout) ───────────────────────────────────
const float RS      = 1.0;
const float B2CRIT  = 6.75;     // (3√3/2)²  — squared critical impact param
const float ISCO    = 3.0;
const float DOUT    = 12.0;
const float ESCR    = 100.0;

// ── Hash (branchless, cheap) ──────────────────────────────────────────────────
float h1(float n){ return fract(sin(n)*43758.545); }
float h2(vec2 p){
    p=fract(p*vec2(443.897,441.423));
    p+=dot(p,p.yx+19.19);
    return fract((p.x+p.y)*p.x);
}
float h3(vec3 p){
    p=fract(p*vec3(443.897,441.423,437.195));
    p+=dot(p,p.zyx+19.19);
    return fract((p.x+p.y)*p.z);
}

// ── ACES tone map ─────────────────────────────────────────────────────────────
vec3 ACES(vec3 x){
    return clamp((x*(2.51*x+.03))/(x*(2.43*x+.59)+.14),0.,1.);
}

// ── Star field — physically motivated stellar colours ─────────────────────────
// Hertzsprung-Russell colour sequence: O=blue-white → M=red-orange
// No purple, no neon. Natural stellar temperature colours only.
vec3 starColor(float t){
    // t in [0,1] from hash — maps to stellar spectral class
    const vec3 M = vec3(1.00,0.55,0.25);  // ~3000 K orange-red
    const vec3 K = vec3(1.00,0.77,0.48);  // ~4500 K warm orange
    const vec3 G = vec3(1.00,0.95,0.80);  // ~5800 K solar white
    const vec3 F = vec3(0.98,0.98,0.95);  // ~7000 K white
    const vec3 A = vec3(0.88,0.92,1.00);  // ~9000 K blue-white
    const vec3 B = vec3(0.70,0.80,1.00);  // ~15000K blue
    if(t<.17) return mix(M,K,t*5.88);
    if(t<.33) return mix(K,G,(t-.17)*6.25);
    if(t<.50) return mix(G,F,(t-.33)*5.88);
    if(t<.67) return mix(F,A,(t-.50)*5.88);
    if(t<.83) return mix(A,B,(t-.67)*6.25);
    return B;
}

vec3 stars(vec3 dir){
    vec3 col=vec3(0);
    float s=40.0;
    for(int k=0;k<5;k++){
        vec3 q=floor(dir*s);
        float hv=h3(q+float(k)*7.3);
        // Step instead of if → no branch divergence
        float bright=step(.9965,hv)*(hv-.9965)*280.0;
        bright*=bright;
        vec3 sc=starColor(h3(q+float(k)*91.3));
        col+=sc*bright;
        s*=2.6;
    }
    return col;
}

// ── Deep-space sky — physically accurate, zero purple ────────────────────────
// Colours sourced from real Milky Way imaging:
//   Disk plane: warm dust-scattered white
//   Core      : warm amber (dust reddening + stellar crowding)
//   Interarm  : extremely dark, faint blue CMB scatter
vec3 sky(vec3 dir){
    // Galactic plane normal (IAU galactic coordinates, approximate)
    const vec3 GN=vec3(0.1946,-0.2458,0.9494);  // galactic north
    const vec3 GC=vec3(-0.0550,-0.8734,0.4840); // galactic centre

    float lat=dot(dir,GN);           // galactic latitude
    float cAng=dot(dir,GC);          // proximity to galactic centre

    // Tight plane glow (exp approximated as clamp for perf)
    float g_tight=max(0.0, 1.0-lat*lat*48.0);       // ~exp(-48x²), fast
    float g_wide =max(0.0, 1.0-lat*lat* 6.5);       // wider halo
    g_tight*=g_tight;  g_wide*=g_wide;               // sharpen falloff

    // Galactic core (avoid expensive acos: use 1-cAng proxy)
    float coreDist=1.0-cAng;
    float g_core=max(0.0, 1.0-coreDist*coreDist*4.5)*g_tight;

    // Very faint large-scale dust mottling (no trig needed)
    float dust=h3(floor(dir*9.0))*0.10+0.92;

    // Physically motivated palette — warm dust, no purple
    vec3 c = vec3(0.003,0.004,0.007)               // deep space ambient
           + vec3(0.24,0.20,0.14)*g_tight*0.55     // warm disc plane
           + vec3(0.18,0.15,0.10)*g_wide *0.10     // extended halo
           + vec3(0.40,0.26,0.12)*g_core *1.60;    // warm amber core

    c*=dust;
    return c + stars(dir);
}

// ── Blackbody colour (no pow, pure mix) ─────────────────────────────────────
// T_norm [0,1] : 0 = cool (orange-red ~3000K), 1 = hot (blue-white ~20000K)
vec3 bbCol(float T){
    T=clamp(T,0.0,1.0);
    const vec3 c0=vec3(0.90,0.32,0.05);  // 3000 K deep orange
    const vec3 c1=vec3(1.00,0.65,0.20);  // 5000 K warm orange
    const vec3 c2=vec3(1.00,0.95,0.80);  // 8000 K warm white
    const vec3 c3=vec3(0.82,0.90,1.00);  // 15000K blue-white
    if(T<.33) return mix(c0,c1,T*3.0);
    if(T<.66) return mix(c1,c2,(T-.33)*3.03);
    return mix(c2,c3,(T-.66)*3.03);
}

// ── Accretion disk — Novikov-Thorne with relativistic Doppler ────────────────
// Optimised: no pow(), no atan(), no log().
// cphi/sphi passed in to allow arm structure without trig calls.
vec4 diskEmit(float r, vec3 er, vec3 ephi, float pr, float L,
              float cphi, float sphi){
    if(r<ISCO||r>DOUT) return vec4(0.0);

    // u = 0 at outer edge, 1 at ISCO
    float u=(DOUT-r)/(DOUT-ISCO);

    // Temperature: Novikov-Thorne T ~ r^(-3/4)
    // Approximation: u^0.65 ≈ √u * (0.79 + 0.21*u)
    // Error < 0.4 % across [0,1]. No pow() needed.
    float sqU =sqrt(u);
    float facU=0.79+0.21*u;
    float Tbase=sqU*facU;

    // Relativistic Keplerian speed (Schwarzschild equatorial)
    // v = sqrt(rs / (2r - 3rs)) [exact for circular orbit]
    float vk=sqrt(RS/max(2.0*r-3.0*RS, 0.15));
    vk=clamp(vk,0.0,0.93);

    // Disk prograde direction (equatorial CCW viewed from +y)
    vec3 pos3=r*er;
    float rxz=max(length(pos3.xz),1e-5);
    vec3 vDir=vec3(-pos3.z,0.0,pos3.x)*inversesqrt(rxz*rxz);

    // Photon direction at crossing
    vec3 pDir=normalize(pr*er+(L/r)*ephi);
    float cosA=dot(pDir,vDir);

    // Doppler factor D = 1 / (γ(1 − v cosα))
    float g2=1.0-vk*vk;
    float gam=inversesqrt(g2);         // γ  (no division)
    float D=1.0/max(gam*(1.0-vk*cosA),0.02);

    // Gravitational redshift √(1 - rs/r)
    float gRed=sqrt(max(1.0-RS/r,0.002));

    // Combined frequency ratio
    float gTot=D*gRed;
    float Tobs=clamp(Tbase*gTot*0.72,0.0,1.0);

    // Bolometric intensity: I ~ D³ × g_red × T^3.5
    // Replace all pow() with multiplications + one sqrt:
    float D2=D*D, D3=D2*D;
    float T2=Tbase*Tbase, T3=T2*Tbase;
    float sqT=sqrt(Tbase);
    float T35=T3*sqT;
    float intense=D3*gRed*T35*3.2;

    // Radial density: peaks near ISCO, zero at outer edge
    float dens=u*max(0.0,1.0-0.75*u);
    dens=clamp(dens*2.5,0.0,1.0);

    // Azimuthal brightness variation using angle-doubling (no atan/trig calls).
    // c2,s2 = cos(2φ),sin(2φ) from tracked (cphi,sphi) — free.
    float c2=cphi*cphi-sphi*sphi;
    float s2=2.0*cphi*sphi;
    // Offset by radius-dependent phase (u as cheap proxy for log r)
    float phaseR=u*2.0;
    float arm=(c2*cos(phaseR)+s2*sin(phaseR))*0.10+0.90;

    vec3 col=bbCol(Tobs);
    return vec4(col*intense*dens*arm, dens*0.85);
}

// ─────────────────────────────────────────────────────────────────────────────
// main
// ─────────────────────────────────────────────────────────────────────────────
void main(){
    vec2 uv=vUV*2.0-1.0;
    uv.x*=aspect;
    vec3 rayDir=normalize(cRight.xyz*(uv.x*fovTan)
                         +cUp.xyz   *(uv.y*fovTan)
                         +cFwd.xyz);

    // ── Orbital plane basis ─────────────────────────────────────────────────
    // Every null geodesic lies in the plane { O, cPos, cPos+rayDir }
    vec3 cam=cPos.xyz;
    vec3 e1=normalize(cam);
    vec3 Np=cross(cam,rayDir);
    float NpL=length(Np);
    vec3 e2;
    if(NpL<1e-5)
        e2=normalize(cross(e1, abs(e1.y)<.9?vec3(0,1,0):vec3(1,0,0)));
    else
        e2=normalize(cross(Np/NpL,e1));

    // ── Initial conditions ──────────────────────────────────────────────────
    float r0 =length(cam);
    float pr =dot(rayDir,e1);   // dr/dλ
    float vt =dot(rayDir,e2);   // tangential
    float L  =r0*vt;            // angular momentum (E=1 → L=b)
    float L2 =L*L;

    // Null condition: pr² + (1−rs/r)(L/r)² = 1
    float f0 =1.0-RS/r0;
    float vr2=1.0-f0*(L/r0)*(L/r0);
    if(vr2<0.0){ FragColor=vec4(0,0,0,1); return; }
    pr=sign(pr)*sqrt(vr2);
    float r=r0;

    // Incremental φ: (cphi,sphi) = (cos φ, sin φ), start at φ=0
    float cphi=1.0,sphi=0.0;

    // ── Integration state ────────────────────────────────────────────────────
    vec4  disk    =vec4(0.0);
    bool  absorbed=false;
    bool  escaped =false;
    float y_prev  =cam.y;       // 3-D world y for disk-crossing detection

    // ─────────────────────────────────────────────────────────────────────────
    // RK4 loop — FIXED NSTEP iterations for Kepler warp coherence.
    // Finished rays use `continue` (fast path, 1 branch) so all threads
    // in a warp still execute the same iteration count → no stall divergence.
    // ─────────────────────────────────────────────────────────────────────────
    for(int i=0;i<NSTEP;i++){

        if(absorbed||escaped) continue;     // fast path for done threads

        // ── Termination checks ────────────────────────────────────────────
        if(r<RS*1.005){ absorbed=true; continue; }
        if(r>ESCR)    { escaped =true; continue; }

        // Below photon sphere, inward, sub-critical → certain capture
        if(r<1.52*RS && pr<0.0 && L2<B2CRIT+0.12){ absorbed=true; continue; }

        // ── Adaptive step size ────────────────────────────────────────────
        // Smooth radial shrink toward horizon + Gaussian at photon sphere
        float u_rs=RS/r;
        float dl=BASE_DL/(1.0+18.0*u_rs/(1.0-u_rs+1e-5));
        float xps=(r-1.5)/1.5;
        dl/=(1.0+9.0*exp(-24.0*xps*xps));

        // ── RK4 on (r, p_r) ──────────────────────────────────────────────
        // Equation is EXACT (not approximate): d(pr)/dλ = L²/r³(1−1.5/r)
        // Derived by substituting null condition into full Christoffel form.
        // Cost ≈ 7 FLOPS per stage vs ~22 for the full 4-variable form.

        float k1r=pr;
        float k1p=L2/(r*r*r)*(1.0-1.5/r);

        float r2=r+.5*dl*k1r, pr2=pr+.5*dl*k1p;
        float k2p=L2/(r2*r2*r2)*(1.0-1.5/r2);

        float r3=r+.5*dl*pr2, pr3=pr+.5*dl*k2p;
        float k3p=L2/(r3*r3*r3)*(1.0-1.5/r3);

        float r4=r+dl*pr3, pr4=pr+dl*k3p;
        float k4p=L2/(r4*r4*r4)*(1.0-1.5/r4);

        float w=dl*(1.0/6.0);
        r +=w*(k1r+2.0*pr2+2.0*pr3+pr4);
        pr+=w*(k1p+2.0*k2p+2.0*k3p+k4p);

        // ── φ update: dφ/dλ = L/r² (exact L conservation) ───────────────
        float dphi=(L/(r*r))*dl;

        // Incremental rotation — 5th-order Taylor for |Δφ|<0.09 (most steps)
        // avoids trig entirely in the inner loop.
        float c_d,s_d;
        if(abs(dphi)<0.09){
            float dp2=dphi*dphi;
            c_d=1.0-dp2*(0.5-dp2*(1.0/24.0));
            s_d=dphi*(1.0-dp2*(1.0/6.0-dp2*(1.0/120.0)));
        } else {
            c_d=cos(dphi); s_d=sin(dphi);
        }
        float nc=cphi*c_d-sphi*s_d;
        float ns=sphi*c_d+cphi*s_d;
        cphi=nc; sphi=ns;

        // Renormalise every 32 steps (prevents (cphi,sphi) length drift)
        if((i&31)==0){
            float inv=inversesqrt(cphi*cphi+sphi*sphi);
            cphi*=inv; sphi*=inv;
        }

        // Null condition re-enforcement every 64 steps
        // Corrects p_r drift; exact L and φ are already maintained above.
        if((i&63)==0 && r>1.1*RS){
            float fc=max(1.0-RS/r,1e-5);
            float vr2c=1.0-fc*(L/r)*(L/r);
            if(vr2c>=0.0) pr=sign(pr)*sqrt(vr2c);
        }

        // ── Accretion disk crossing ───────────────────────────────────────
        // Disk lies in world y = 0 (equatorial plane), rs < r < D_OUT
        float y_cur=r*(cphi*e1.y+sphi*e2.y);

        if(y_prev*y_cur<0.0 && disk.a<0.97){
            if(r>=ISCO && r<=DOUT){
                vec3 er_c =cphi*e1+sphi*e2;
                vec3 ep_c =-sphi*e1+cphi*e2;
                vec4 dc=diskEmit(r,er_c,ep_c,pr,L,cphi,sphi);
                // Front-to-back alpha compositing (supports multiple crossings)
                disk.rgb+=(1.0-disk.a)*dc.a*dc.rgb;
                disk.a  +=(1.0-disk.a)*dc.a;
            }
        }
        y_prev=y_cur;
    }

    // ── Post-integration compositing ─────────────────────────────────────────
    vec3 final;

    if(absorbed){
        final=vec3(0.0);    // event horizon: genuine black
    } else {
        // Escape direction in 3-D world space
        vec3 erf = cphi*e1+sphi*e2;
        vec3 epf =-sphi*e1+cphi*e2;
        vec3 escDir=normalize(pr*erf+(L/r)*epf);

        // Sample sky at escape direction
        final=sky(escDir);

        // Gravitational blueshift (photons gain energy falling to observer)
        // Factor = 1/sqrt(1−rs/r0); cap to avoid over-brightening near BH
        float gb=1.0/sqrt(max(1.0-RS/r0,0.60));
        final*=min(gb,1.35);
    }

    // Composite disk over sky / black
    final=mix(final,disk.rgb,clamp(disk.a,0.0,1.0));

    // ── Tone mapping + gamma ──────────────────────────────────────────────────
    // ACES filmic preserves highlights while lifting shadows naturally
    final=ACES(final*1.15);
    final=pow(max(final,vec3(0.0)),vec3(1.0/2.2));

    FragColor=vec4(final,1.0);
}
)GLSL";

// ─────────────────────────────────────────────────────────────────────────────
// Camera + motion detection
// ─────────────────────────────────────────────────────────────────────────────
struct Camera {
    float r     = 15.0f;
    float theta = float(M_PI)*0.38f;
    float phi   = 0.0f;
    double lx=0,ly=0;
    bool   drag=false;
    double lastMoveT = -99.0;   // for dynamic scaling

    vec3 pos()  const { return vec3(r*sinf(theta)*cosf(phi),
                                    r*cosf(theta),
                                    r*sinf(theta)*sinf(phi)); }
    vec3 fwd()  const { return normalize(-pos()); }
    vec3 right()const { return normalize(cross(fwd(),vec3(0,1,0))); }
    vec3 up()   const { return normalize(cross(right(),fwd())); }
    float fovTan()const{ return tanf(radians(65.0f)*0.5f); }
    bool  moving(double now)const{ return now-lastMoveT < 0.20; }
} gCam;
int gW=1280, gH=720;

// ─────────────────────────────────────────────────────────────────────────────
// FBO for dynamic render scaling
// ─────────────────────────────────────────────────────────────────────────────
static GLuint gFBO=0, gFBOtex=0;
static int    gFBOw=0, gFBOh=0;

static void ensureFBO(int w,int h){
    if(w==gFBOw&&h==gFBOh) return;
    if(gFBOtex) glDeleteTextures(1,&gFBOtex);
    if(gFBO)    glDeleteFramebuffers(1,&gFBO);
    glGenTextures(1,&gFBOtex);
    glBindTexture(GL_TEXTURE_2D,gFBOtex);
    glTexImage2D(GL_TEXTURE_2D,0,GL_RGB16F,w,h,0,GL_RGB,GL_HALF_FLOAT,nullptr);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_MIN_FILTER,GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_MAG_FILTER,GL_LINEAR);
    glGenFramebuffers(1,&gFBO);
    glBindFramebuffer(GL_FRAMEBUFFER,gFBO);
    glFramebufferTexture2D(GL_FRAMEBUFFER,GL_COLOR_ATTACHMENT0,GL_TEXTURE_2D,gFBOtex,0);
    glBindFramebuffer(GL_FRAMEBUFFER,0);
    gFBOw=w; gFBOh=h;
}

// ─────────────────────────────────────────────────────────────────────────────
// GL helpers
// ─────────────────────────────────────────────────────────────────────────────
static GLuint compileShader(GLenum type,const char* src){
    GLuint s=glCreateShader(type);
    glShaderSource(s,1,&src,nullptr);
    glCompileShader(s);
    GLint ok; glGetShaderiv(s,GL_COMPILE_STATUS,&ok);
    if(!ok){
        char buf[4096]; glGetShaderInfoLog(s,4096,nullptr,buf);
        std::cerr<<(type==GL_VERTEX_SHADER?"VS":"FS")<<" error:\n"<<buf<<"\n";
        return 0;
    }
    return s;
}
static GLuint linkProg(GLuint vs,GLuint fs){
    GLuint p=glCreateProgram();
    glAttachShader(p,vs); glAttachShader(p,fs);
    glLinkProgram(p);
    GLint ok; glGetProgramiv(p,GL_LINK_STATUS,&ok);
    if(!ok){
        char buf[4096]; glGetProgramInfoLog(p,4096,nullptr,buf);
        std::cerr<<"Link:\n"<<buf<<"\n"; return 0;
    }
    glDeleteShader(vs); glDeleteShader(fs);
    return p;
}

static void cbResize(GLFWwindow*,int w,int h){
    gW=w; gH=h; glViewport(0,0,w,h);
}
static void cbScroll(GLFWwindow*w,double,double dy){
    auto*c=(Camera*)glfwGetWindowUserPointer(w);
    c->r=std::clamp(c->r*(dy>0?.92f:1.09f),2.1f,45.0f);
    c->lastMoveT=glfwGetTime();
}
static void cbMouse(GLFWwindow*w,int btn,int act,int){
    auto*c=(Camera*)glfwGetWindowUserPointer(w);
    if(btn==GLFW_MOUSE_BUTTON_LEFT) c->drag=(act==GLFW_PRESS);
    if(act==GLFW_PRESS) glfwGetCursorPos(w,&c->lx,&c->ly);
}
static void cbCursor(GLFWwindow*w,double x,double y){
    auto*c=(Camera*)glfwGetWindowUserPointer(w);
    if(c->drag){
        c->phi  -=float(x-c->lx)*.005f;
        c->theta =std::clamp(c->theta-float(y-c->ly)*.005f,.03f,float(M_PI)-.03f);
        c->lastMoveT=glfwGetTime();
    }
    c->lx=x; c->ly=y;
}
static void cbKey(GLFWwindow*w,int key,int,int act,int){
    if(act!=GLFW_PRESS) return;
    auto*c=(Camera*)glfwGetWindowUserPointer(w);
    if(key==GLFW_KEY_R){ c->r=15;c->theta=float(M_PI)*.38f;c->phi=0;c->lastMoveT=glfwGetTime(); }
    if(key==GLFW_KEY_ESCAPE) glfwSetWindowShouldClose(w,1);
}

    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR,4);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR,3);
    glfwWindowHint(GLFW_OPENGL_PROFILE,GLFW_OPENGL_CORE_PROFILE);

    GLFWwindow*win=glfwCreateWindow(gW,gH,
        "Schwarzschild BH · OpenGL 4.3 · Quadro K2100M",nullptr,nullptr);
    if(!win){ fputs("Window\n",stderr); glfwTerminate(); return 1; }
    glfwMakeContextCurrent(win);
    glfwSwapInterval(0);   // uncapped for FPS measurement; set to 1 to cap at 60

    glewExperimental=GL_TRUE;
    if(glewInit()!=GLEW_OK){ fputs("GLEW\n",stderr); return 1; }
    printf("OpenGL  : %s\nRenderer: %s\n\n",
           glGetString(GL_VERSION),glGetString(GL_RENDERER));

    // Callbacks
    glfwSetWindowUserPointer(win,&gCam);
    glfwSetFramebufferSizeCallback(win,cbResize);
    glfwSetScrollCallback(win,cbScroll);
    glfwSetMouseButtonCallback(win,cbMouse);
    glfwSetCursorPosCallback(win,cbCursor);
    glfwSetKeyCallback(win,cbKey);

    // Fullscreen quad (triangle strip)
    float quad[]={ -1,-1, 1,-1, -1,1, 1,1 };
    GLuint vao,vbo;
    glGenVertexArrays(1,&vao); glGenBuffers(1,&vbo);
    glBindVertexArray(vao);
    glBindBuffer(GL_ARRAY_BUFFER,vbo);
    glBufferData(GL_ARRAY_BUFFER,sizeof(quad),quad,GL_STATIC_DRAW);
    glVertexAttribPointer(0,2,GL_FLOAT,GL_FALSE,0,nullptr);
    glEnableVertexAttribArray(0);
    glBindVertexArray(0);

    // Camera UBO
    GLuint ubo;
    glGenBuffers(1,&ubo);
    glBindBuffer(GL_UNIFORM_BUFFER,ubo);
    glBufferData(GL_UNIFORM_BUFFER,sizeof(CamUBO),nullptr,GL_DYNAMIC_DRAW);
    glBindBufferBase(GL_UNIFORM_BUFFER,0,ubo);

    // Compile
    GLuint prog=linkProg(compileShader(GL_VERTEX_SHADER,VERT),
                         compileShader(GL_FRAGMENT_SHADER,FRAG));
    if(!prog){ glfwTerminate(); return 1; }
    puts("Shaders compiled OK.");

    double fpsT=glfwGetTime(); int fpsN=0;

    while(!glfwWindowShouldClose(win)){
        double t=glfwGetTime();
        bool mv=gCam.moving(t);

        // Dynamic render scale: ½ resolution while camera is moving
        float scale=mv ? 0.5f : 1.0f;
        int rW=std::max(1,(int)(gW*scale));
        int rH=std::max(1,(int)(gH*scale));

        // Ensure FBO matches current render resolution
        ensureFBO(rW,rH);

        // Upload camera UBO
        {
            CamUBO d;
            d.pos  =vec4(gCam.pos(),  0);
            d.right=vec4(gCam.right(),0);
            d.up   =vec4(gCam.up(),   0);
            d.fwd  =vec4(gCam.fwd(),  0);
            d.fovTan=gCam.fovTan();
            d.aspect=float(rW)/float(rH);
            d.time  =float(t);
            d.pad   =0;
            glBindBuffer(GL_UNIFORM_BUFFER,ubo);
            glBufferSubData(GL_UNIFORM_BUFFER,0,sizeof(d),&d);
        }

        // Render to FBO at (possibly reduced) resolution
        glBindFramebuffer(GL_FRAMEBUFFER,gFBO);
        glViewport(0,0,rW,rH);
        glUseProgram(prog);
        glBindVertexArray(vao);
        glDrawArrays(GL_TRIANGLE_STRIP,0,4);

        // Blit from FBO to default framebuffer (bilinear upscale if ½ res)
        glBindFramebuffer(GL_READ_FRAMEBUFFER,gFBO);
        glBindFramebuffer(GL_DRAW_FRAMEBUFFER,0);
        glViewport(0,0,gW,gH);
        glBlitFramebuffer(0,0,rW,rH,0,0,gW,gH,
                          GL_COLOR_BUFFER_BIT,GL_LINEAR);
        glBindFramebuffer(GL_FRAMEBUFFER,0);

        glfwSwapBuffers(win);
        glfwPollEvents();

        // FPS display
        fpsN++;
        if(t-fpsT>=1.0){
            printf("\rFPS:%3d  r=%.1frs  θ=%.0f°  %s        ",
                   fpsN, gCam.r,
                   gCam.theta*180.f/float(M_PI),
                   mv?"[½ res]":"[full  ]");
            fflush(stdout);
            fpsN=0; fpsT=t;
        }
    }

    glfwDestroyWindow(win);
    glfwTerminate();
    return 0;
}
